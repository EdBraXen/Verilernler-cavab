    }

public static int correctSolution(int[][] tree) {
    // tree[i] = {leftChild, rightChild}, index 1-based
    boolean[] isRoot = new boolean[tree.length];
    Arrays.fill(isRoot, true);

    // Identify the root (node that is never a child)
    for (int i = 1; i < tree.length; i++) {
        int left = tree[i][0];
        int right = tree[i][1];
        if (left != -1) isRoot[left] = false;
        if (right != -1) isRoot[right] = false;
    }

    int root = 1;
    for (int i = 1; i < isRoot.length; i++) {
        if (isRoot[i]) {
            root = i;
            break;
        }
    }

    return height(root, tree);
}

private static int height(int node, int[][] tree) {
    if (node == -1) return -1;
    int left = tree[node][0];
    int right = tree[node][1];
    return Math.max(height(left, tree), height(right, tree)) + 1;
}

        Stack<Integer> s1 = new Stack<>(), s2 = new Stack<>();
        List<Integer> out = new ArrayList<>();

            String[] p = op.split(" ");
            switch (p[0]) {
                case "1" -> s1.push(Integer.parseInt(p[1]));

                    if (s2.isEmpty()) while (!s1.isEmpty()) s2.push(s1.pop());

                }
            }
        }

public static long correctSolution(int[] arr) {
        int n = arr.length;
        long comparisons = 0;
        
        for (int i = 0; i < n - 1; i++) {
            int maxIdx = i;
            for (int j = i + 1; j < n; j++) {
                comparisons++; // Hər müqayisəni sayırıq
                if (arr[j] > arr[maxIdx]) {
                    maxIdx = j;
                }
            }
            // Yerdəyişmə (Swap)
            int temp = arr[maxIdx];
            arr[maxIdx] = arr[i];
            arr[i] = temp;
        }
        return comparisons;
    }

        }

public static int[] correctSolution(int[] arr, int x) {
    int n = arr.length;
    int first = -1, last = -1;

    // Find first occurrence
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] >= x) right = mid - 1;
        else left = mid + 1;
    }
    if (left < n && arr[left] == x) first = left;

    // Find last occurrence
    left = 0; right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] <= x) left = mid + 1;
        else right = mid - 1;
    }
    if (right >= 0 && arr[right] == x) last = right;

    return new int[]{first, last};
}

public static int correctSolution(int n, int[][] edges) {
    List<List<Integer>> graph = new ArrayList<>();
    for (int i = 0; i <= n; i++) graph.add(new ArrayList<>());
    for (int[] edge : edges) {
        int u = edge[0], v = edge[1];
        graph.get(u).add(v);
        graph.get(v).add(u);
    }
    boolean[] visited = new boolean[n + 1];
    int count = 0;
    for (int i = 1; i <= n; i++) {
        if (!visited[i]) {
            dfs(i, graph, visited);
            count++;
        }
    }
    return count;
}

private static void dfs(int node, List<List<Integer>> graph, boolean[] visited) {
    visited[node] = true;
    for (int neighbor : graph.get(node)) {
        if (!visited[neighbor]) dfs(neighbor, graph, visited);
    }
}

public static int correctSolution(int[] priorities, int m) {
    Queue<int[]> queue = new LinkedList<>();
    for (int i = 0; i < priorities.length; i++) {
        queue.offer(new int[]{priorities[i], i});
    }

    int printed = 0;

    while (!queue.isEmpty()) {
        int[] current = queue.poll();
        int priority = current[0];
        boolean hasHigher = false;

        for (int[] doc : queue) {
            if (doc[0] > priority) {
                hasHigher = true;
                break;
            }
        }

        if (hasHigher) {
            queue.offer(current); // geri göndər
        } else {
            printed++;
            if (current[1] == m) return printed;
        }
    }
    return -1;
}

public static String correctSolution(String cardNumber) {
    int len = cardNumber.length();
    if (len < 16) return "";

    // İlk 12 rəqəmi maskala
    String maskedPart = "";
    for (int i = 0; i < 12; i++) {
        maskedPart += "*";
    }

    // Son 4 rəqəm
    String last4 = cardNumber.substring(len - 4);

    // Birləşdirilmiş maskalanmış string
    String combined = maskedPart + last4;

    // 4-lük qruplara böl və tire əlavə et
    String result = "";
    for (int i = 0; i < combined.length(); i++) {
        if (i > 0 && i % 4 == 0) {
            result += "-";
        }
        result += combined.charAt(i);
    }

    return result;
}

  if (board == null) {
        board = new char[n][n];
        for (int i = 0; i < n; i++) Arrays.fill(board[i], '.');
    }

    // Baza halı: Bütün sətirlər tamamlanıbsa, lövhəni string siyahısına çevir
    if (row == n) {
        List<String> currentBoard = new ArrayList<>();
        for (char[] r : board) currentBoard.add(new String(r));
        results.add(currentBoard);
        return results;
    }

    for (int col = 0; col < n; col++) {
        boolean safe = true;
        // Təhlükəsizlik yoxlanışı (Sütun və Diaqonallar)
        for (int i = 0; i < row; i++) {
            if (board[i][col] == 'Q') safe = false; // Sütun
            int colDiff = row - i;
            if (col - colDiff >= 0 && board[i][col - colDiff] == 'Q') safe = false; // Sol diaqonal
            if (col + colDiff < n && board[i][col + colDiff] == 'Q') safe = false;  // Sağ diaqonal
            if (!safe) break;
        }

        if (safe) {
            board[row][col] = 'Q';
            results.addAll(correctSolution(n, row + 1, board)); // Rekursiv çağırış
            board[row][col] = '.'; // Backtrack
        }
    }
    return results;
}

public static List<Integer> correctSolution(int[][] matrix) {
    List<Integer> result = new ArrayList<>();
    if (matrix == null || matrix.length == 0) return result;

    int top = 0, bottom = matrix.length - 1;
    int left = 0, right = matrix[0].length - 1;

    while (top <= bottom && left <= right) {
        // Right: Left to Right
        for (int i = left; i <= right; i++) result.add(matrix[top][i]);
        top++;

        // Down: Top to Bottom
        for (int i = top; i <= bottom; i++) result.add(matrix[i][right]);
        right--;

        if (top <= bottom) {
            // Left: Right to Left
            for (int i = right; i >= left; i--) result.add(matrix[bottom][i]);
            bottom--;
        }

        if (left <= right) {
            // Up: Bottom to Top
            for (int i = bottom; i >= top; i--) result.add(matrix[i][left]);
            left++;
        }
    }
    return result;
}

        if (s.charAt(i) == s.charAt(i - 1)) {
